<script>
let data = JSON.parse(localStorage.getItem("rpg")) || {
    You: {xp:0},
    Friend: {xp:0}
};

function level(xp){
    return Math.floor(xp / 100) + 1;
}

const subjects = {
    "Physics": ["Motion Awakening","Force Trials","Energy Rise"],
    "Chemistry": ["Element Origins","Atomic Structure","Bonding Forces"],
    "Math": ["Algebra Trials","Geometry Land","Calculus Rise"],
    "Biology": ["Cell Awakening","Genetics Gate","Human System"],
    "ICT": ["Digital Awakening","Programming Path","Database Dungeon"]
};

const container = document.getElementById("game");

function render(){
    container.innerHTML = "";

    for(let subject in subjects){
        let card = document.createElement("div");
        card.className = "card";

        let title = document.createElement("h3");
        title.innerText = subject;
        card.appendChild(title);

        subjects[subject].forEach(q => {
            let div = document.createElement("div");
            div.className = "quest";
            div.innerHTML = `
                ${q}
                <button onclick="complete('You')">You +10 XP</button>
                <button onclick="complete('Friend')">Friend +10 XP</button>
            `;
            card.appendChild(div);
        });

        container.appendChild(card);
    }

    updateStats();
}

function complete(player){
    data[player].xp += 10;
    localStorage.setItem("rpg", JSON.stringify(data));
    updateStats();
}

function updateStats(){
    document.getElementById("xp1").innerText = data.You.xp;
    document.getElementById("xp2").innerText = data.Friend.xp;

    document.getElementById("lvl1").innerText = level(data.You.xp);
    document.getElementById("lvl2").innerText = level(data.Friend.xp);
}

render();
</script>
